; Test pentru pointeri în NanoASM v2.0
print "Test Pointeri:"
newline

; Alocăm 64 de octeți folosind malloc (SYSCALL_MALLOC = 2)
mov rax, 2          ; Numărul syscall-ului pentru malloc
mov rdi, 64         ; Cerem 64 de octeți
int 0x80            ; În rax primim adresa pointerului alocat!

; Să presupunem că salvăm adresa într-un registru de bază, de ex. rsi
; Dar ca să fie simplu, hai să folosim direct valoarea din rax.
; (Notă: în Nano OS, poți folosi rdi ca pointer).

print "Memorie alocata cu succes!"
newline
exit
